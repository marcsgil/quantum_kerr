\section{Spatially Structured Bosonic Field}

Consider a bosonic quantum field $\mathcal{E}(\mathbf{r})$ satisfying the commutation relations
\begin{equation}
    \label{eq:commutation_relations}
    \comm{\mathcal{E}(\mathbf{r}_1)}{\mathcal{E}^\dagger(\mathbf{r}_2)} = \delta^{(2)}(\mathbf{r}_1 - \mathbf{r}_2), \ \comm{\mathcal{E}(\mathbf{r}_1)}{\mathcal{E}(\mathbf{r}_2)} = \comm{\mathcal{E}^\dagger(\mathbf{r}_1)}{\mathcal{E}^\dagger(\mathbf{r}_2)} = 0\ 
\end{equation}
In the above equation and in what follows, $\mathbf{r} = (x, y)$ is the transverse coordinate.

One can introduce ladder operators $a_v, a_v^\dagger$ associated with the spatial mode $v$ by the expression
\begin{equation}
    a(v) = \int_{\mathbb{R}^2} v^*(\mathbf{r}) \mathcal{E}(\mathbf{r}) d\mathbf{r}.
\end{equation}

We then have
\begin{equation}
    \comm{a(v_1)}{a(v_2)^\dagger} = \int_{\mathbb{R}^2} v_1^*(\mathbf{r}) v_2(\mathbf{r}) d\mathbf{r} =\braket{v_1}{v_2},
\end{equation}
while $\comm{a(v_1)}{a(v_2)} = \comm{a(v_1)^\dagger}{a(v_2)^\dagger} = 0$.

Let us define the quadrature associated to the spatial mode $v$ as
\begin{equation}
    \begin{aligned}
        X(v) &= \frac{1}{\sqrt{2}} \int_{\mathbb{R}^2} v^*(\mathbf{r}) \mathcal{E}(\mathbf{r}) + v(\mathbf{r})\mathcal{E}^\dagger(\mathbf{r}) d\mathbf{r} \\ &= \frac{a(v) + a(v)^\dagger}{\sqrt{2}}
    \end{aligned}
\end{equation}
We then have the commutation relation
\begin{equation}
    \comm{X(v_1)}{X(v_2)} = i \Im \braket{v_1}{v_2},
\end{equation}
which leads to the uncertainty principle
\begin{equation}
    \Delta X(v_1)^2 \Delta X(v_2)^2 \ge \frac{\abs{\Im \braket{v_1}{v_2}}^2}{4}.
\end{equation}
One could also introduce the complementary quadrature $P(v) = X(-iv)$ so that
\begin{equation}
    \comm{X_{v_1}}{P_{v_2}} = i \Re \braket{v_1}{v_2}
\end{equation}
and
\begin{equation}
    \Delta X(v_1)^2 \Delta P(v_2)^2 \ge \frac{\abs{\Re \braket{v_1}{v_2}}^2}{4}.
\end{equation}

We are interested in observables of the form
\begin{equation}
    \label{eq:quadrature_correlation}
    C(v_1, v_2) = \frac{1}{2}\expval{\acomm{X(v_1)}{X(v_2)}} - \expval{X(v_1)}\expval{X(v_2)} =  \Re\left[ \frac{\braket{v_1}{v_2}}{2} + f(v_1, v_2) + g(v_1, v_2) \right],
\end{equation}
where we have defined
\begin{equation}
    \begin{aligned}
        f(v_1, v_2) &= \expval{a(v_1)a(v_2)} - \expval{a(v_1)} \expval{a(v_2)} \\
        g(v_1, v_2) &= \expval{a(v_2)^\dagger a(v_1)} - \expval{a(v_2)^\dagger} \expval{a(v_1)} 
    \end{aligned} .
\end{equation}

In terms of the fields, the required quantities are written as
\begin{equation}
    f(v_1, v_2) = \int v_1^*(\mathbf{r}_2) v_2^*(\mathbf{r}_1) F(\mathbf{r}_1, \mathbf{r}_2) d\mathbf{r}_1 d\mathbf{r}_2; \ \ \ F(\mathbf{r}_1, \mathbf{r}_2) = \expval{\mathcal{E}(\mathbf{r}_1, z) \mathcal{E}(\mathbf{r}_2, z)} - \expval{\mathcal{E}(\mathbf{r}_1)}\expval{\mathcal{E}(\mathbf{r}_2)}
\end{equation}
\begin{equation}
    g(v_1, v_2) = \int v_1^*(\mathbf{r}_1) v_2(\mathbf{r}_2) G(\mathbf{r}_1, \mathbf{r}_2) d\mathbf{r}_1 d\mathbf{r}_2; \ \ \ G(\mathbf{r}_1, \mathbf{r}_2) = \expval{\mathcal{E}^\dagger(\mathbf{r}_2, z) \mathcal{E}(\mathbf{r}_1, z)} - \expval{\mathcal{E}^\dagger(\mathbf{r}_2, z)} \expval{\mathcal{E}(\mathbf{r}_1, z)}
\end{equation}

This formulation is useful because $f$ and $g$ are normally ordered expectation values, which can be directly obtained from stochastic simulations in the positive P representation. Furthermore, this representation also makes it easier to perform linear combinations of the modes, which will be useful later. These combinations will be based on the properties
\begin{equation}
    \begin{aligned}
        f(v_1, v_2 + \lambda v_3) &= f(v_1, v_2) + \lambda^* f(v_1, v_3) \\
        f(v_1, v_2) &= f(v_2, v_1) \\
        g(v_1, v_2 + \lambda v_3) &= g(v_1, v_2) + \lambda g(v_1, v_3) \\
        g(v_1, v_2) &= g(v_2, v_1)^*
    \end{aligned}
\end{equation}
which are a reflection of the fact that $a(v)$ is antilinear while $a^\dagger(v)$ is linear.

\subsection{One mode correlation}

Let us analyze the correlation matrix $\sigma_{jk} = \frac{1}{2}\expval{\acomm{Z_j}{Z_k}} - \expval{Z_j}\expval{Z_k}$ where $\mathbf{Z} = \left( X(v), P(v) \right)$. We observe that $g(v,v) \in \mathbb{R}$, so we get
\begin{equation}
    \sigma = \begin{pmatrix}
        \frac{1}{2} + g(v,v) & 0 \\
        0 & \frac{1}{2} + g(v,v)
    \end{pmatrix} + \begin{pmatrix}
        \Re f(v, v) & -\Im f(v,v) \\
        -\Im f(v,v) & -\Re f(v, v)
    \end{pmatrix} .
\end{equation}

Notice that $f(e^{i\phi}v, e^{i\phi}v) = e^{-i2\phi} f(v, v)$. Therefore, we can diagonalize $\sigma$ by choosing $\phi = [\arg f(v, v) \pm \pi]/ 2$. We then get eigenvalues
\begin{equation}
    \lambda_{\pm} = \frac{1}{2} + g(v, v) \pm \abs{f(v, v)} .
\end{equation}
We then see that the quadrature rotated along the axis $\phi$ has minimum variance of $\lambda_-$, while the one rotated along the axis $\phi + \pi/2$ has maximum variance $\lambda_+$. Furthermore, we have squeezing if $\abs{f(v, v)} > g(v, v)$.

\subsection{Duan Criterion}

Consider two orthonormal states $v_1, v_2$, i.e., $\braket{v_j}{v_k} = \delta_{jk}$. The Duan criterion \cite{duan2000inseparability} states that if
\begin{equation}
    D = \expval{\Delta \left[X(v_1) + X(v_2)\right]^2} + \expval{\Delta\left[P(v_1) - P(v_2)\right]^2} < 2,
\end{equation}
then the states are entangled.

Let us define $v_\pm = v_1 \pm v_2$. We then have
\begin{equation}
    D = \expval{\Delta X(v_+)^2} + \expval{\Delta P(v_-)^2}.
\end{equation}
Although $v_{\pm}$ is not normalized, \eqref{eq:quadrature_correlation} still holds. We then obtain
\begin{equation}
    \begin{aligned}
        D &= 2 + g(v_+, v_+) + g(v_-, v_-) + \Re \left[f(v_+, v_+) - f(v_-, v_-) \right] \\ 
        &= 2\left[1 + g(v_1, v_1) + g(v_2, v_2)\right] + 4 \Re f(v_1, v_2)
    \end{aligned}
\end{equation}

Once again, we can maximize the possible violation by rotating the quadrature of the modes, say, $v_2$ by an angle $\phi = \left[\arg f(v_1, v_2) \pm \pi\right] / 2$. We get an optimal possible violation of
\begin{equation}
    D_{\text{opt}} = 2\left[1 + g(v_1, v_1) + g(v_2, v_2)\right] - 4 \abs{f(v_1, v_2)}
\end{equation}


\section{The Kerr System}

\subsection{Heisenberg Equation for the Field and Linear Solution}

We will assume that the field is propagating along the $z$ direction, and satisfies the Heisenberg equation of motion
\begin{equation}
    \label{eq:heisenberg_equation}
    2ik \partial_z \mathcal{E}(\mathbf{r}, z) + \nabla^2 \mathcal{E}(\mathbf{r}, z) = g \mathcal{E}^\dagger(\mathbf{r}, z) \mathcal{E}^2 (\mathbf{r}, z).
\end{equation}
Let us decompose $\mathcal{E} = \mathcal{E}_0 + \delta\mathcal{E}$ where $\mathcal{E}_0$ is the solution of
\begin{equation}
    2ik \partial_z \mathcal{E}_0(\mathbf{r}, z) + \nabla^2 \mathcal{E}_0(\mathbf{r}, z) = 0, \ \ \ \mathcal{E}_0(\mathbf{r}, 0) = \mathcal{E}(\mathbf{r}, 0).
\end{equation}
Let $u$ be a solution of the paraxial wave equation $2ik \partial_z u(\mathbf{r}, z) + \nabla^2 u(\mathbf{r}, z) = 0$. Then, the anihilation operator corresponding to mode $u$ is given by
\begin{equation}
    a(u, z) = \int u^*(\mathbf{r}, z) \mathcal{E}(\mathbf{r}, z) d\mathbf{r} = a_0(u) + \delta a(u, z)
\end{equation}
where 
\begin{equation}
    a_0(u) = \int u^*(\mathbf{r}, z) \mathcal{E}_0(\mathbf{r}, z) d\mathbf{r}
\end{equation}
is independent of $z$ (take a $z$ derivative, use de equations of motion and integrate by parts) and
\begin{equation}
    \delta a(u, z) = \int u^*(\mathbf{r}, z) \delta\mathcal{E}(\mathbf{r}, z) d\mathbf{r}    
\end{equation}

Our initial state will be a coherent one:
\begin{equation}
    \ket{\alpha, u} = D(\alpha, u) \ket{0} = \exp\left(\alpha a_0(u)^\dagger - \alpha^* a_0(u)\right) \ket{0}.
\end{equation}
This state is an eigenvector of the free field:
\begin{equation}
    \mathcal{E}_0(\mathbf{r}, z) \ket{\alpha, u} = \alpha u(\mathbf{r}, z) \ket{\alpha, u}
\end{equation}

The variation $\delta \mathcal{E}$ satisfies
\begin{equation}
    2ik \partial_z \delta \mathcal{E}(\mathbf{r}, z) + \nabla^2 \delta\mathcal{E}(\mathbf{r}, z) = g \mathcal{E}^\dagger(\mathbf{r}, z) \mathcal{E}^2 (\mathbf{r}, z).
\end{equation}
Evaluating this at $z=0$ gives us
\begin{equation}
    2ik \partial_z \delta \mathcal{E}(\mathbf{r}, 0) = g \mathcal{E}_0^\dagger(\mathbf{r}, 0) \mathcal{E}^2_0 (\mathbf{r}, 0)
\end{equation}
from which we obtain the Taylor expansion
\begin{equation}
    \delta \mathcal{E}(\mathbf{r}, z) \approx -\frac{igz}{2k} \mathcal{E}_0^\dagger(\mathbf{r}, 0) \mathcal{E}^2_0 (\mathbf{r}, 0)
\end{equation}

For the complete field we then have
\begin{equation}
    \mathcal{E}(\mathbf{r}, z) \approx \mathcal{E}_0(\mathbf{r}, 0) - \frac{igz}{2k}\mathcal{E}_0^\dagger(\mathbf{r}, 0) \mathcal{E}^2_0 (\mathbf{r}, 0)
\end{equation}
This expression will be the basis of our further developments.

\subsection{Expectation value of field}

Taking expectation values, we arrive at
\begin{equation}
    \label{eq:exp_val_field}
    \expval{\mathcal{E}(\mathbf{r}, z)} \approx \alpha u(\mathbf{r}, 0) - \frac{igz \abs{\alpha}^2 \alpha}{2k} \abs{u(\mathbf{r}, 0)}^2 u(\mathbf{r}, 0) .
\end{equation}
The expectation value of the annihilation operator $a_v$ corresponding to a spatial mode $v$ is then
\begin{equation}
    \label{eq:expval_anihilation}
    \expval{a_v} = \alpha \braket{v}{u}  - \frac{igz \abs{\alpha}^2 \alpha}{2k} \braket{v, u}{u, u}
\end{equation}
In the above expression we have introduced the second spatial overlap
\begin{equation}
    \braket{v_1, v_2}{u_1, u_2} = \int v_1^*(\mathbf{r},z) v_2^*(\mathbf{r},z) u_1(\mathbf{r},z) u_2(\mathbf{r},z) d\mathbf{r} .
\end{equation}

When $u = u_{0l}$ is a Laguerre-Gaussian mode with radial order $p=0$ and topological charge $l$, we have that
\begin{equation}
    \abs{u_{0l}(\mathbf{r}, z)} u_{0l}^2(\mathbf{r}, z) = \sum_{q=0}^{\abs{l}} C_{ql} \overline{u}_{ql}(\mathbf{r}, z)
\end{equation}
where $\overline{u}_{ql}$ are Laguerre-Gaussian modes with a reduced waist. Therefore, one may write
\begin{equation}
    \expval{a_v} = \alpha \braket{v}{u_{0,l}}  - \frac{igz \abs{\alpha}^2 \alpha}{2k} \sum_{q=0}^{\abs{l}} C_{ql} \braket{v}{\overline{u}_{ql}}
\end{equation}

In particular, when $v = \overline{u}_{ql}$, we the have that
\begin{equation}
    \expval{a_{\overline{u}_{ql}}} = \begin{cases}
        \alpha \braket{\overline{u}_{ql}}{u_{0,l}} - \dfrac{igz \abs{\alpha}^2 \alpha C_{ql}}{2k} & q \le \abs{l} \\
        \alpha \braket{\overline{u}_{ql}}{u_{0,l}}  & q > \abs{l}
    \end{cases} .
\end{equation}

\subsection{Correlation functions}

One of the two point field function is
\begin{equation}
    \mathcal{E}^\dagger(\mathbf{r}_1, z) \mathcal{E}(\mathbf{r}_2, z) \approx \mathcal{E}_0^\dagger(\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z) + \mathcal{E}_0^\dagger(\mathbf{r}_1, z) \delta\mathcal{E}(\mathbf{r}_2, z) + \delta\mathcal{E}^\dagger(\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z)
\end{equation}
where we have only kept terms linear in $\delta \mathcal{E}$.

Close to $z = 0$, we have that
\begin{equation}
    \begin{aligned}
        \mathcal{E}^\dagger(\mathbf{r}_1, z) \mathcal{E}(\mathbf{r}_2, z) \approx \mathcal{E}_0^\dagger(\mathbf{r}_1, 0) \mathcal{E}_0(\mathbf{r}_2, 0) - \frac{igz}{2k} \left[ \mathcal{E}_0^\dagger(\mathbf{r}_1, 0) \mathcal{E}_0^\dagger(\mathbf{r}_2, 0) \mathcal{E}^2_0 (\mathbf{r}_2, 0) - \mathcal{E}_0^\dagger(\mathbf{r}_1, 0)^2 \mathcal{E}_0 (\mathbf{r}_1, 0) \mathcal{E}_0 (\mathbf{r}_2, 0) \right]
    \end{aligned}
\end{equation}
Evaluating this at our initial state gives us
\begin{equation}
    \begin{aligned}
        \expval{\mathcal{E}^\dagger(\mathbf{r}_1, z) \mathcal{E}(\mathbf{r}_2, z)} &\approx \abs{\alpha}^2 u^*(\mathbf{r}_1,0) u(\mathbf{r}_2, 0)\left\{ 1 - \frac{igz\abs{\alpha}^2}{2k} \left[ \abs{u(\mathbf{r}_2, 0)}^2 - \abs{u(\mathbf{r}_1, 0)}^2 \right]   \right\} \\ &\approx \expval{\mathcal{E}^\dagger(\mathbf{r}_1, z)} \expval{\mathcal{E}(\mathbf{r}_2, z)}
    \end{aligned}
\end{equation}
where the last approximate inequality comes from direct comparison with \eqref{eq:exp_val_field}. In particular, this shows that, for any $v_1$ and $v_2$, we have $G(\mathbf{r}_1, \mathbf{r}_2) \approx 0$.

We can also use this to write the expectation value of the number operator:
\begin{equation}
    \expval{n(v)} = \expval{a^\dagger(v) a(v)} \approx \abs{\expval{a(v)}}^2 \approx \abs{\alpha \braket{v}{u}}^2 + \frac{gz \abs{\alpha}^4}{k} \Im \braket{v}{u} \braket{u, u}{u,v} 
\end{equation}
When both $u$ and $v$ are arbitrary Laguerre-Gaussian modes (any $p,l,w$), we have that  $\Im \braket{v}{u} \braket{u, u}{u,v} = 0$, so there is no first order correction to the occupation number.

The other two point field function is
\begin{equation}
    \begin{aligned}
        \mathcal{E}(\mathbf{r}_1, z) \mathcal{E}(\mathbf{r}_2, z) &\approx \mathcal{E}_0(\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z) + \mathcal{E}_0(\mathbf{r}_1, z) \delta\mathcal{E}(\mathbf{r}_2, z) + \delta\mathcal{E}(\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z) \\ &\approx \mathcal{E}_0(\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z) - \frac{igz}{2k} \left[ \mathcal{E}_0(\mathbf{r}_1, z)\mathcal{E}_0^\dagger(\mathbf{r}_2, z) \mathcal{E}^2_0 (\mathbf{r}_2, z) + \mathcal{E}_0^\dagger(\mathbf{r}_1, z) \mathcal{E}^2_0 (\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z) \right] \\ &= \mathcal{E}_0(\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z) - \frac{igz}{2k} \left[ \mathcal{E}_0^\dagger(\mathbf{r}_2, z) \mathcal{E}_0(\mathbf{r}_1, z)\mathcal{E}^2_0 (\mathbf{r}_2, z) + \mathcal{E}_0^\dagger(\mathbf{r}_1, z) \mathcal{E}^2_0 (\mathbf{r}_1, z) \mathcal{E}_0(\mathbf{r}_2, z) + \delta(\mathbf{r}_1 - \mathbf{r}_2) \mathcal{E}^2_0 (\mathbf{r}_2, z) \right]
    \end{aligned}
\end{equation}

The appearance of the delta function here is important, because now
\begin{equation}
    \begin{aligned}
        \expval{\mathcal{E}(\mathbf{r}_1, z) \mathcal{E}(\mathbf{r}_2, z)} &\approx \alpha^2u(\mathbf{r}_1,z) u(\mathbf{r}_2,z) \left[ 1 - \frac{igz \abs{\alpha}^2}{2k} \left[ \abs{u(\mathbf{r}_2, z)}^2 + \abs{u(\mathbf{r}_1, z)}^2 \right] \right] - \frac{igz \alpha^2}{2k} \delta(\mathbf{r}_1 - \mathbf{r}_2) u^2 (\mathbf{r}_2, z) \\ & \approx \expval{\mathcal{E}(\mathbf{r}_1, z)} \expval{\mathcal{E}(\mathbf{r}_2, z)} - \frac{igz \alpha^2}{2k} \delta(\mathbf{r}_1 - \mathbf{r}_2) u^2 (\mathbf{r}_2, z)
    \end{aligned}
\end{equation}

Therefore, we may conclude that
\begin{equation}
    F(\mathbf{r}_1, \mathbf{r}_2) \approx -\frac{igz \alpha^2}{2k} \delta(\mathbf{r}_1 - \mathbf{r}_2) u^2 (\mathbf{r}_2, z)
\end{equation}
and
\begin{equation}
    f(v_1, v_2) \approx -\frac{igz \alpha^2}{2k} \braket{v_1,v_2}{u,u} .
\end{equation}

This leads us to a minimum quadrature variance for mode $v$ of
\begin{equation}
    \lambda_- = \frac{1}{2} - \frac{z}{2k} \abs{g\alpha^2\braket{v,v}{u,u}}
\end{equation}
and an optimal Duan parameter for modes $v_1$ and $v_2$ of
\begin{equation}
    D_{\text{opt}} = 2 - \frac{2z}{k} \abs{g\alpha^2\braket{v_1,v_2}{u,u}}.
\end{equation}
The corresponding angles are $\phi_{\text{squeezing}} = (-\pi/2 \pm \pi + \arg g\alpha^2\braket{v,v}{u,u}) / 2$ and $\phi_{\text{duan}} = (-\pi/2 \pm \pi + \arg g\alpha^2\braket{v_1,v_2}{u,u}) / 2$

At this point, it becomes clear that the overlaps of the form $\braket{v_1,v_2}{u,u}$ are the ones that dominate this sort of process. On the other hand, the generation of radial orders comes from the properties of the different overlap $\braket{v,u}{u,u}$, which appears in \eqref{eq:expval_anihilation}. It is therefore hard to see any special implication of the selection rules here.

Consider the case where all the involved modes are vortices: $u(\mathbf{r}) = u_r(r)e^{il_u\phi}$ and similarly for the $v$ functions. We have squeezing if $l_v = l_u$ and we have entanglement if $l_{v_1} + l_{v_2} = 2l_u$. If all the radial functions are positive, as is the case for $LG_{0,0}$ and $LG_{0,\pm1}$, and $\alpha$ is positive, then the optimal angles are both $\text{sign}(g)\pi / 4$. 


\section{Numerical Considerations}

Note that, according to \eqref{eq:commutation_relations}, the field $\mathcal{E}(\mathbf{r})$ has dimensions of $1/\sqrt{\text{area}}$. Therefore, by dimensional analysis on eq. \eqref{eq:heisenberg_equation}, we have that $g$ is dimensionless. 

In order to put the equation in a more convenient form, we introduce the dimensionless coordinate $\tilde{\mathbf{r}} = \mathbf{r}/w_0$ where $w_0$ is the waist of the input beam. We also introduce the dimensionless propagation distance $\tilde{z} = z/z_R$ where $z_R = kw_0^2/2$ is the Rayleigh range. Finally, we introduce the rescaled field $\tilde{\mathcal{E}}(\tilde{\mathbf{r}}, \tilde{z}) = \sqrt{\Delta A} \mathcal{E}(\mathbf{r}, z)$, where $\Delta A$ is the area of a pixel in the numerical grid. In terms of these variables, the commutation relations read
\begin{equation}
    \comm{\tilde{\mathcal{E}}(\tilde{\mathbf{r}})}{\tilde{\mathcal{E}}(\tilde{\mathbf{r}})^\dagger} \approx \delta_{\tilde{\mathbf{r}}, \tilde{\mathbf{r}}'},
\end{equation}
while the equation of motion reads
\begin{equation}
    i \partial_{\tilde{z}} \tilde{\mathcal{E}}(\tilde{\mathbf{r}}, \tilde{z}) = - \frac{1}{4} \tilde{\nabla}^2 \tilde{\mathcal{E}}(\tilde{\mathbf{r}}, \tilde{z}) +\frac{g}{4 \Delta \tilde{A}} \tilde{\mathcal{E}}^\dagger(\tilde{\mathbf{r}}, \tilde{z}) \tilde{\mathcal{E}}^2 (\tilde{\mathbf{r}}, \tilde{z})
\end{equation}
where $\Delta \tilde{A} = \Delta A/w_0^2$ is the area of a pixel in the rescaled coordinates.

In the positive $P$ representation, the field $\tilde{\mathcal{E}}$ is represented by two independent complex stochastic fields $\psi_1$ and $\psi_2$ satisfying the stochastic equations (compare with \cite{PhysRevLett.98.120402})
\begin{equation}
    \begin{aligned}
        i \partial_{\tilde{z}} \psi_1(\tilde{\mathbf{r}}, \tilde{z}) &= - \frac{1}{4} \tilde{\nabla}^2 \psi_1(\tilde{\mathbf{r}}, \tilde{z}) + \frac{g}{4 \Delta \tilde{A}} \psi_2(\tilde{\mathbf{r  }}, \tilde{z}) \psi_1^2 (\tilde{\mathbf{r}}, \tilde{z}) + \sqrt{\frac{ig}{4\Delta \tilde{A}}} \psi_1(\tilde{\mathbf{r}}, \tilde{z}) \xi_1(\tilde{\mathbf{r}}, \tilde{z}) \\
        -i \partial_{\tilde{z}} \psi_2(\tilde{\mathbf{r}}, \tilde{z}) &= - \frac{1}{4} \tilde{\nabla}^2 \psi_2(\tilde{\mathbf{r}}, \tilde{z}) + \frac{g}{4 \Delta \tilde{A}} \psi_1(\tilde{\mathbf{r}}, \tilde{z}) \psi_2^2 (\tilde{\mathbf{r}}, \tilde{z}) + \sqrt{-\frac{ig}{4\Delta \tilde{A}}} \psi_2(\tilde{\mathbf{r}}, \tilde{z}) \xi_2(\tilde{\mathbf{r}}, \tilde{z})
    \end{aligned}
\end{equation}
where $\xi_1$ and $\xi_2$ are independent real Gaussian white noises with zero mean and correlations
\begin{equation}
    \expval{\xi_j(\tilde{\mathbf{r}}, \tilde{z}) \xi_k(\tilde{\mathbf{r}}', \tilde{z}')} = \delta_{jk} \delta_{\tilde{\mathbf{r}}, \tilde{\mathbf{r}}'} \delta_{\tilde{z}, \tilde{z}'}
\end{equation}
For a coherent initial state in the mode $u$, we have that $\psi_1(\tilde{\mathbf{r}}, 0) = \alpha u(\tilde{\mathbf{r}}, 0)$ and $\psi_2(\tilde{\mathbf{r}}, 0) = \alpha^* u^*(\tilde{\mathbf{r}}, 0)$.
